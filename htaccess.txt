# --- Redirection canonique : force https:// et www. ---
# Évite que Google indexe 4 versions différentes du site
# (http, https, sans www, avec www) et disperse le référencement.
RewriteEngine On

RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} !^www\. [NC]
RewriteCond %{HTTP_HOST} ^(?:www\.)?(.+)$ [NC]
RewriteRule ^ https://www.%1%{REQUEST_URI} [L,R=301]

# --- Page 404 personnalisée ---
ErrorDocument 404 /404.html

# --- Limite d'upload pour la pièce jointe du formulaire de contact (5 Mo) ---
# La valeur par défaut de PHP (souvent 2 Mo) est trop basse pour le formulaire de contact.
<IfModule mod_php.c>
    php_value upload_max_filesize 8M
    php_value post_max_size 10M
</IfModule>

# --- Compression Gzip (page plus légère à transférer) ---
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml
    AddOutputFilterByType DEFLATE application/javascript application/json
    AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>

# --- Cache navigateur (les visiteurs qui reviennent chargent le site plus vite) ---
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType image/webp    "access plus 1 year"
    ExpiresByType image/jpeg    "access plus 1 year"
    ExpiresByType image/png    "access plus 1 year"
    ExpiresByType image/svg+xml "access plus 1 year"
    ExpiresByType text/css     "access plus 1 month"
    ExpiresByType application/javascript "access plus 1 month"
</IfModule>

# --- Sécurité de base ---
<IfModule mod_headers.c>
    Header set X-Content-Type-Options "nosniff"
    Header set X-Frame-Options "SAMEORIGIN"
    Header set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

# --- Empêche l'accès direct aux fichiers sensibles ---
<FilesMatch "^(composer\.json|composer\.lock|cleanup-images\.sh)$">
    Require all denied
</FilesMatch>
